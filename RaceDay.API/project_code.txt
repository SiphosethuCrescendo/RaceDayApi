using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using RaceDay.Core.Entities;
using RaceDay.Infrastructure.Data;

namespace RaceDay.API.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class CategoriesController : ControllerBase
    {
        private readonly RaceDayDbContext _context;
        public CategoriesController(RaceDayDbContext context) { _context = context; }

        [HttpGet("event/{eventId}")]
        [AllowAnonymous]
        public async Task<IActionResult> ForEvent(int eventId)
        {
            var list = await _context.Categories.Where(c => c.EventID == eventId).ToListAsync();
            return Ok(list);
        }

        [HttpPost]
        [Authorize(Roles = "Organiser")]
        public async Task<IActionResult> Create([FromBody] Category category)
        {
            category.CategoryID = 0;
            _context.Categories.Add(category);
            await _context.SaveChangesAsync();
            return Ok(category);
        }

        [HttpDelete("{id}")]
        [Authorize(Roles = "Organiser")]
        public async Task<IActionResult> Delete(int id)
        {
            var c = await _context.Categories.FindAsync(id);
            if (c == null) return NotFound(new { message = "Category not found" });
            _context.Categories.Remove(c);
            await _context.SaveChangesAsync();
            return Ok(new { message = "Category deleted" });
        }
    }
}

        }

        [Authorize(Roles = "Organiser")]
        public async Task<IActionResult> Update(int id, [FromBody] UpdateEventDto dto)
        {
            try
            {
                Event updated = await _eventService.UpdateAsync(id, dto, CurrentUserId);
                if (updated == null)
                {
                    return NotFound(new { message = "Event not found or you do not own it" });
                }
                return Ok(updated);
            }
            catch (Exception ex)
            {
                return BadRequest(new { message = ex.Message });
            }
        }

        [HttpDelete("{id}")]
        [Authorize(Roles = "Organiser")]
        public async Task<IActionResult> Delete(int id)
        {
            bool deleted = await _eventService.DeleteAsync(id, CurrentUserId);
            if (deleted == false)
            {
                return NotFound(new { message = "Event not found or you do not own it" });
            }
            return Ok(new { message = "Event deleted" });
        }

        [HttpPost("upload-image")]
        [Authorize(Roles = "Organiser")]

using System;
using System.Collections.Generic;
using System.Security.Claims;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using RaceDay.Core.DTOs;
using RaceDay.Core.Entities;
using RaceDay.Infrastructure.Data;

namespace RaceDay.API.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    [Authorize]
    public class ResultsController : ControllerBase
    {
        private readonly RaceDayDbContext _context;

        public ResultsController(RaceDayDbContext context)
        {
            _context = context;
        }

        private int CurrentUserId
        {
            get
            {
                var idClaim = User.FindFirst(ClaimTypes.NameIdentifier);
                if (idClaim == null)
                {
                    return 0;
                }
                return int.Parse(idClaim.Value);
            }
        }

        // POST: /api/results  (Organiser owner only)
        [HttpPost]
        [Authorize(Roles = "Organiser")]
        public async Task<IActionResult> Upload([FromBody] CreateResultDto dto)
        {
            // Find the enrolment
            Enrolment enrolment = await _context.Enrolments
                .Include(e => e.Category)
                    .ThenInclude(c => c.Event)
                .FirstOrDefaultAsync(e => e.EnrolmentID == dto.EnrolmentID);

            if (enrolment == null)
            {
                return NotFound(new { message = "Enrolment not found" });
            }

            // Check organiser owns the event
            if (enrolment.Category.Event.OrganiserID != CurrentUserId)
            {
                return Forbid();
            }

            // Check a result doesn't already exist
            Result existing = await _context.Results
                .FirstOrDefaultAsync(r => r.EnrolmentID == dto.EnrolmentID);

            if (existing != null)
            {
                return BadRequest(new { message = "A result already exists for this enrolment" });
            }

            // Create the result
            Result newResult = new Result();
            newResult.EnrolmentID = dto.EnrolmentID;
            newResult.FinishTime = dto.FinishTime;
            newResult.Position = dto.Position;
            newResult.GunTime = dto.GunTime;
            newResult.ChipTime = dto.ChipTime;
            newResult.Status = "Valid";

            _context.Results.Add(newResult);

            // Mark enrolment as Completed
            enrolment.Status = "Completed";

            await _context.SaveChangesAsync();

            return Ok(newResult);
        }

        // GET: /api/results/my  (Participant only)
        [HttpGet("my")]
        [Authorize(Roles = "Participant")]
        public async Task<IActionResult> MyResults()
        {
            List<Result> results = await _context.Results
                .Include(r => r.Enrolment)
                    .ThenInclude(e => e.Category)
                        .ThenInclude(c => c.Event)
                .Where(r => r.Enrolment.ParticipantID == CurrentUserId)
                .ToListAsync();

            return Ok(results);
        }

        // GET: /api/results/event/{eventId}  (public)
        [HttpGet("event/{eventId}")]
        [AllowAnonymous]
        public async Task<IActionResult> EventResults(int eventId)
        {
            List<Result> results = await _context.Results
                .Include(r => r.Enrolment)
                    .ThenInclude(e => e.Category)
                .Where(r => r.Enrolment.Category.EventID == eventId)
                .ToListAsync();

            return Ok(results);
        }
    }
}

        {
            _context = context;
        }

        {
            get
            {
                var idClaim = User.FindFirst(ClaimTypes.NameIdentifier);
                if (idClaim == null)
                {
                    return 0;

                return int.Parse(idClaim.Value);
            }
        }

        // GET: /api/users/profile
        [HttpGet("profile")]
        public async Task<IActionResult> GetProfile()

                .FirstOrDefaultAsync(u => u.UserID == CurrentUserId);

            if (user == null)
            {
                return NotFound(new { message = "User not found" });

            dto.UserID = user.UserID;
            dto.Email = user.Email;
            dto.FullName = user.FullName;

        [HttpPut("profile")]

        {
            User user = await _context.Users
                .FirstOrDefaultAsync(u => u.UserID == CurrentUserId);

            if (user == null)
            {
                return NotFound(new { message = "User not found" });
            }

            user.FullName = dto.FullName;
            user.PhoneNumber = dto.PhoneNumber;

            await _context.SaveChangesAsync();

            return Ok(new { message = "Profile updated" });
        }

using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.EntityFrameworkCore;
using Microsoft.IdentityModel.Tokens;
using Microsoft.OpenApi.Models;
using RaceDay.Core.Interfaces;
using RaceDay.Infrastructure.Data;
using RaceDay.Infrastructure.Services;
using System.Text;
using System.Text.Json.Serialization;

var builder = WebApplication.CreateBuilder(args);

// 1. Database Connection
builder.Services.AddDbContext<RaceDayDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));

// 2. Register Services
builder.Services.AddScoped<IAuthService, AuthService>();
builder.Services.AddScoped<IEventService, EventService>();
builder.Services.AddScoped<IEnrolmentService, EnrolmentService>();

builder.Services.AddControllers()
    .AddJsonOptions(o => o.JsonSerializerOptions.ReferenceHandler = ReferenceHandler.IgnoreCycles);

// 3. JWT Authentication
string jwtKey = builder.Configuration["Jwt:Key"] ?? "";
string jwtIssuer = builder.Configuration["Jwt:Issuer"] ?? "";
string jwtAudience = builder.Configuration["Jwt:Audience"] ?? "";

builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidateAudience = true,
            ValidateLifetime = true,
            ValidateIssuerSigningKey = true,
            ValidIssuer = jwtIssuer,
            ValidAudience = jwtAudience,
            IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtKey))
        };
    });

builder.Services.AddAuthorization();

// 4. Swagger
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen(options =>
{
    options.SwaggerDoc("v1", new OpenApiInfo { Title = "RaceDay API", Version = "v1" });

    options.AddSecurityDefinition("Bearer", new OpenApiSecurityScheme
    {
        Name = "Authorization",
        Type = SecuritySchemeType.Http,
        Scheme = "Bearer",
        BearerFormat = "JWT",
        In = ParameterLocation.Header,
        Description = "Paste the JWT token here."
    });

    options.AddSecurityRequirement(new OpenApiSecurityRequirement
    {
        {
            new OpenApiSecurityScheme
            {
                Reference = new OpenApiReference { Type = ReferenceType.SecurityScheme, Id = "Bearer" }
            },
            new string[] { }
        }
    });
});

var app = builder.Build();

// 5. HTTP Pipeline
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();

using (var scope = app.Services.CreateScope())
{
    scope.ServiceProvider.GetRequiredService<RaceDayDbContext>().Database.Migrate();
}

app.Run();
